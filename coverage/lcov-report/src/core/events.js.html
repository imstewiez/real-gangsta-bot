const EVENT_NAMES = Object.keys(EVENTS);
 
/**
 * Verifica se um evento pertence ao catálogo formal.
 * @param {string} name — nome do evento (e.g. 'saida.closed')
 * @returns {boolean}
 */
function isKnownEvent(name) {
  return name in EVENTS;
}
